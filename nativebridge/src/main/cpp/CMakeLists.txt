cmake_minimum_required(VERSION 3.22.1)
project(launchapk_bridge LANGUAGES C CXX)
set_property(GLOBAL PROPERTY JOB_POOLS launchapk_compile=2)
set(CMAKE_JOB_POOL_COMPILE launchapk_compile)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
set(NOEGL ON CACHE BOOL "" FORCE)
set(NOX11 ON CACHE BOOL "" FORCE)
set(GBM OFF CACHE BOOL "" FORCE)
set(USE_ANDROID_LOG ON CACHE BOOL "" FORCE)
set(USE_CCACHE OFF CACHE BOOL "" FORCE)
set(NO_INIT_CONSTRUCTOR ON CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../renderer/third_party/gl4es" "${CMAKE_CURRENT_BINARY_DIR}/gl4es")
set_target_properties(GL PROPERTIES OUTPUT_NAME "gl4es" SUFFIX ".so" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_LIBRARY_OUTPUT_DIRECTORY}")
target_link_libraries(GL m dl log)
set(THIRD_PARTY "${CMAKE_CURRENT_SOURCE_DIR}/../../../third_party")
foreach(option UTILS EXAMPLES INSTALL INSTALL_CONFIG INSTALL_HRTF_DATA INSTALL_AMBDEC_PRESETS INSTALL_EXAMPLES INSTALL_UTILS UPDATE_BUILD_VERSION)
    set(ALSOFT_${option} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(backend PIPEWIRE PULSEAUDIO ALSA OSS SOLARIS SNDIO JACK COREAUDIO OBOE PORTAUDIO SDL2 WAVE)
    set(ALSOFT_BACKEND_${backend} OFF CACHE BOOL "" FORCE)
endforeach()
set(ALSOFT_BACKEND_OPENSL ON CACHE BOOL "" FORCE)
set(ALSOFT_REQUIRE_OPENSL ON CACHE BOOL "" FORCE)
set(LIBTYPE SHARED CACHE STRING "" FORCE)
add_subdirectory("${THIRD_PARTY}/openal-soft" "${CMAKE_CURRENT_BINARY_DIR}/openal-soft")
set_target_properties(OpenAL PROPERTIES OUTPUT_NAME "openal" VERSION "" SOVERSION "")
add_library(glfw SHARED android_glfw.cpp)
target_include_directories(glfw PRIVATE "${THIRD_PARTY}/glfw/include")
target_compile_options(glfw PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation)
target_link_libraries(glfw PRIVATE android EGL dl log)
add_library(launchapk_bridge SHARED bridge.cpp jvm_launch.cpp game_jni.cpp)
target_compile_options(launchapk_bridge PRIVATE -Wall -Wextra -Werror)
target_link_libraries(launchapk_bridge PRIVATE glfw android EGL dl log)
add_dependencies(launchapk_bridge GL)
# Device-side native test executable; EXCLUDE_FROM_ALL keeps tests out of the APK.
add_executable(launchapk_events_test EXCLUDE_FROM_ALL "${CMAKE_CURRENT_SOURCE_DIR}/../../test/cpp/game_events_test.cpp")
target_include_directories(launchapk_events_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(launchapk_events_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
